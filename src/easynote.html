(function () { "use strict"; const STORAGE_KEY = "easynote.cards.v1"; /** @type {{id:string, title:string, body:string, tags:string[], createdAt:number, updatedAt:number}[]} */ let notes = []; let activeTag = "__all__"; let searchTerm = ""; // ---------- Elements ---------- const cardGrid = document.getElementById("cardGrid"); const tagList = document.getElementById("tagList"); const searchInput = document.getElementById("searchInput"); const viewTitle = document.getElementById("viewTitle"); const viewSub = document.getElementById("viewSub"); const noResults = document.getElementById("noResults"); const emptyHint = document.getElementById("emptyHint"); const countAll = document.getElementById("countAll"); const overlay = document.getElementById("overlay"); const noteForm = document.getElementById("noteForm"); const editId = document.getElementById("editId"); const titleInput = document.getElementById("titleInput"); const bodyInput = document.getElementById("bodyInput"); const tagInput = document.getElementById("tagInput"); const newCardBtn = document.getElementById("newCardBtn"); const cancelBtn = document.getElementById("cancelBtn"); const confirmOverlay = document.getElementById("confirmOverlay"); const confirmDelete = document.getElementById("confirmDelete"); const confirmCancel = document.getElementById("confirmCancel"); let pendingDeleteId = null; // ---------- Storage ---------- function load() { try { const raw = localStorage.getItem(STORAGE_KEY); notes = raw ? "All cards" : `Tagged “${activeTag}”`; viewSub.textContent = `${list.length} note${list.length === 1 ? "" : "s"}`; if (list.length === 0) { cardGrid.innerHTML = ""; noResults.classList.remove("hidden"); noResults.querySelector("p").textContent = notes.length === 0 ? "No cards yet. Click “+ New card” to write your first note." : "No cards match that search."; return; } noResults.classList.add("hidden"); cardGrid.innerHTML = list .map((n) => { const primaryTag = n.tags[0]; return `
${primaryTag ? `${escapeHtml(primaryTag)}` : ""}

${escapeHtml(n.title)}

${escapeHtml(n.body)}

${ n.tags.length ? `
${n.tags.map((t) => `#${escapeHtml(t)}`).join("")}
` : "" }
${formatDate(n.updatedAt)}
`; }) .join(""); cardGrid.querySelectorAll(".edit-btn").forEach((btn) => { btn.addEventListener("click", (e) => { e.stopPropagation(); const id = btn.closest(".index-card").dataset.id; openForm(id); }); }); cardGrid.querySelectorAll(".delete-btn").forEach((btn) => { btn.addEventListener("click", (e) => { e.stopPropagation(); const id = btn.closest(".index-card").dataset.id; pendingDeleteId = id; confirmOverlay.classList.remove("hidden"); }); }); cardGrid.querySelectorAll(".index-card").forEach((card) => { card.addEventListener("click", () => openForm(card.dataset.id)); }); } // ---------- Form (add/edit) ---------- function openForm(id) { noteForm.reset(); if (id) { const note = notes.find((n) => n.id === id); if (!note) return; editId.value = note.id; titleInput.value = note.title; bodyInput.value = note.body; tagInput.value = note.tags.join(", "); } else { editId.value = ""; } overlay.classList.remove("hidden"); setTimeout(() => titleInput.focus(), 50); } function closeForm() { overlay.classList.add("hidden"); noteForm.reset(); editId.value = ""; } noteForm.addEventListener("submit", (e) => { e.preventDefault(); const title = titleInput.value.trim(); const body = bodyInput.value.trim(); const tags = parseTags(tagInput.value); if (!title || !body) return; const now = Date.now(); if (editId.value) { const note = notes.find((n) => n.id === editId.value); closeForm(); render(); }); newCardBtn.addEventListener("click", () => openForm(null)); cancelBtn.addEventListener("click", closeForm); overlay.addEventListener("click", (e) => { if (e.target === overlay) closeForm(); pendingDeleteId = null; } } }); // ---------- Delete confirm ---------- confirmDelete.addEventListener("click", () => { if (pendingDeleteId) { notes = notes.filter((n) => n.id !== pendingDeleteId); save(); // if the deleted note's tag no longer exists and it was active, reset filter if (activeTag !== "__all__" && !allTags().includes(activeTag)) { activeTag = "__all__"; } render(); } pendingDeleteId = null; confirmOverlay.classList.add("hidden"); }); confirmCancel.addEventListener("click", () => { pendingDeleteId = null; confirmOverlay.classList.add("hidden"); }); confirmOverlay.addEventListener("click", (e) => { if (e.target === confirmOverlay) { pendingDeleteId = null; confirmOverlay.classList.add("hidden"); } }); // ---------- Search ---------- searchInput.addEventListener("input", (e) => { searchTerm = e.target.value; renderCards(); }); // ---------- Init ---------- load(); render(); })();
